cmake_minimum_required(VERSION 3.24)
project(flymsg_native LANGUAGES CXX)
add_executable(flymsg_native main.cpp)
target_compile_features(flymsg_native PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(flymsg_native PRIVATE /fp:strict /W4)
else()
  target_compile_options(flymsg_native PRIVATE -ffp-contract=off -Wall -Wextra -Werror)
endif()
if(WIN32)
  target_sources(flymsg_native PRIVATE dx12.cpp)
  target_compile_definitions(flymsg_native PRIVATE FLYMSG_DX12 NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(flymsg_native PRIVATE d3d12 dxgi)
endif()
